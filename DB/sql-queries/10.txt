Task
-----------
Für welche Produkte gibt es ähnliche Produkte in einer anderen Hauptkategorie? 

Hinweis: Eine Hauptkategorie ist eine Produktkategorie ohne Oberkategorie.
         Erstellen Sie eine rekursive Anfrage, die zu jedem Produkt dessen Hauptkategorie bestimmt.



Query
-----------
WITH RECURSIVE topdown_kategorie(hauptkategorie_id, kategorie_id) AS (
    SELECT
        k.kategorie_id AS hauptkategorie_id,
        k.kategorie_id AS kategorie_id
    FROM kategorie k
    WHERE k.parent_kategorie IS NULL

    UNION ALL

    SELECT
        tk.hauptkategorie_id,
        k.kategorie_id
    FROM topdown_kategorie tk
    JOIN kategorie k ON k.parent_kategorie = tk.kategorie_id
),

produkt_hauptkategorie AS (
    SELECT pk.produkt_id, tk.hauptkategorie_id
    FROM produktkategorie pk
    JOIN topdown_kategorie tk ON pk.kategorie_id = tk.kategorie_id
),

related_products AS (
	SELECT a.produkt_ida as produkt_id
	FROM aehnlichzu a
	WHERE EXISTS (
		SELECT 1
		FROM produkt_hauptkategorie pkb
		WHERE pkb.produkt_id = a.produkt_idb AND NOT EXISTS (
			SELECT 1
			FROM produkt_hauptkategorie pka
			WHERE pka.produkt_id = a.produkt_ida AND pkb.hauptkategorie_id = pka.hauptkategorie_id
		)
	)
	UNION
	SELECT a.produkt_idb as produkt_id
	FROM aehnlichzu a
	WHERE EXISTS (
		SELECT 1
		FROM produkt_hauptkategorie pka
		WHERE pka.produkt_id = a.produkt_ida AND NOT EXISTS (
			SELECT 1
			FROM produkt_hauptkategorie pkb
			WHERE pkb.produkt_id = a.produkt_idb AND pka.hauptkategorie_id = pkb.hauptkategorie_id
		)
	)
)

SELECT * FROM related_products;
